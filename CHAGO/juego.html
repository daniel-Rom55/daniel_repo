import math import random import pygame pygame.init() pygame.mouse.set_visible(False) pygame.event.set_grab(True) WIDTH, HEIGHT = 1280, 720 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Zombie Forest 3D") clock = pygame.time.Clock() MAP = [ "111111111111111", "100000000000001", "101111011111101", "100100000000001", "101101111011101", "100001000010001", "101101011110101", "100000010000001", "101111010111101", "100000000000001", "101011111011101", "100010000010001", "101110111110101", "100000000000001", "111111111111111", ] FOV = math.radians(70) MAX_DEPTH = 18.0 weapons = { "pistol": {"damage": 35, "cooldown": 0.38, "reload": 1.1, "mag_size": 12}, "rifle": {"damage": 18, "cooldown": 0.10, "reload": 1.7, "mag_size": 30}, } player = { "x": 2.5, "y": 2.5, "angle": 0.0, "health": 100.0, "weapon": "pistol", "ammo": { "pistol": {"mag": 12, "reserve": 48}, "rifle": {"mag": 30, "reserve": 90}, }, "reload_timer": 0.0, "last_shot": 0.0, "score": 0, } wave = 1 zombies = [] respawn_timer = 0.0 game_over = False def clamp(v, lo, hi): return max(lo, min(v, hi)) def normalize_angle(a): while a < -math.pi: a += 2 * math.pi while a > math.pi: a -= 2 * math.pi return a def is_wall(x, y): if x < 0 or y < 0: return True ix = int(x) iy = int(y) if iy < 0 or iy >= len(MAP): return True if ix < 0 or ix >= len(MAP[0]): return True return MAP[iy][ix] == "1" def cast_ray(angle): dist = 0.0 while dist < MAX_DEPTH: x = player["x"] + math.cos(angle) * dist y = player["y"] + math.sin(angle) * dist if is_wall(x, y): return dist, x, y dist += 0.02 return MAX_DEPTH, None, None def spawn_zombies(count): items = [] for _ in range(count): tries = 0 while tries < 200: x = random.uniform(1.5, 13.5) y = random.uniform(1.5, 13.5) if abs(x - player["x"]) > 3.0 and abs(y - player["y"]) > 3.0 and not is_wall(x, y): items.append({ "x": x, "y": y, "hp": 100, "speed": random.uniform(0.8, 1.5), "alive": True, "attack_cooldown": 0.0, "hit_flash": 0.0, }) break tries += 1 return items def reset_game(): global zombies, wave, respawn_timer, game_over player["x"] = 2.5 player["y"] = 2.5 player["angle"] = 0.0 player["health"] = 100.0 player["weapon"] = "pistol" player["score"] = 0 player["reload_timer"] = 0.0 player["last_shot"] = 0.0 for w in weapons: player["ammo"][w]["mag"] = weapons[w]["mag_size"] player["ammo"][w]["reserve"] = 48 if w == "pistol" else 90 wave = 1 zombies = spawn_zombies(5) respawn_timer = 0.0 game_over = False def reload_weapon(): if player["reload_timer"] > 0: return w = player["weapon"] ammo = player["ammo"][w] if ammo["mag"] >= weapons[w]["mag_size"]: return if ammo["reserve"] <= 0: return player["reload_timer"] = weapons[w]["reload"] def finish_reload(): w = player["weapon"] ammo = player["ammo"][w] needed = weapons[w]["mag_size"] - ammo["mag"] if needed > 0 and ammo["reserve"] > 0: taken = min(needed, ammo["reserve"]) ammo["mag"] += taken ammo["reserve"] -= taken def can_shoot(): now = pygame.time.get_ticks() / 1000.0 return now - player["last_shot"] >= weapons[player["weapon"]]["cooldown"] def fire(): global game_over if game_over: return w = player["weapon"] ammo = player["ammo"][w] if player["reload_timer"] > 0: return if ammo["mag"] <= 0: reload_weapon() return if not can_shoot(): return ammo["mag"] -= 1 player["last_shot"] = pygame.time.get_ticks() / 1000.0 mouse_x, mouse_y = pygame.mouse.get_pos() aim_x = (mouse_x - WIDTH / 2) / WIDTH aim_y = (mouse_y - HEIGHT / 2) / HEIGHT aim_angle = player["angle"] + aim_x * FOV * 1.3 best_target = None best_diff = 999.0 for z in zombies: if not z["alive"]: continue dx = z["x"] - player["x"] dy = z["y"] - player["y"] dist = math.hypot(dx, dy) if dist > 8.0: continue ang = math.atan2(dy, dx) diff = abs(normalize_angle(ang - aim_angle)) if diff < best_diff: best_target = z best_diff = diff if best_target is not None and best_diff < 0.14: best_target["hp"] -= weapons[w]["damage"] best_target["hit_flash"] = 0.18 if best_target["hp"] <= 0: best_target["alive"] = False player["score"] += 10 if not any(z["alive"] for z in zombies): return def update_player(dt, keys): global game_over if game_over: return move = 0.0 strafe = 0.0 if keys[pygame.K_w]: move += 1.0 if keys[pygame.K_s]: move -= 1.0 if keys[pygame.K_d]: strafe += 1.0 if keys[pygame.K_a]: strafe -= 1.0 speed = 2.9 * dt forward_x = math.cos(player["angle"]) * move * speed forward_y = math.sin(player["angle"]) * move * speed side_x = math.cos(player["angle"] + math.pi / 2) * strafe * speed side_y = math.sin(player["angle"] + math.pi / 2) * strafe * speed nx = player["x"] + forward_x + side_x ny = player["y"] + forward_y + side_y if not is_wall(nx, player["y"]): player["x"] = nx if not is_wall(player["x"], ny): player["y"] = ny rel = pygame.mouse.get_rel() player["angle"] += rel[0] * 0.0032 if player["reload_timer"] > 0: player["reload_timer"] -= dt if player["reload_timer"] <= 0: finish_reload() def update_zombies(dt): global respawn_timer, wave, game_over for z in zombies: if not z["alive"]: continue z["hit_flash"] = max(0.0, z["hit_flash"] - dt) dx = player["x"] - z["x"] dy = player["y"] - z["y"] dist = math.hypot(dx, dy) if dist < 0.001: dist = 0.001 if dist > 1.0: step = z["speed"] * dt nx = z["x"] + (dx / dist) * step ny = z["y"] + (dy / dist) * step if not is_wall(nx, z["y"]): z["x"] = nx if not is_wall(z["x"], ny): z["y"] = ny if dist < 1.1: z["attack_cooldown"] -= dt if z["attack_cooldown"] <= 0: player["health"] -= 13.0 z["attack_cooldown"] = 0.9 if not any(z["alive"] for z in zombies): if respawn_timer <= 0: respawn_timer = 2.5 respawn_timer -= dt if respawn_timer <= 0: wave += 1 zombies.extend(spawn_zombies(4 + wave)) respawn_timer = 0.0 if player["health"] <= 0: game_over = True def draw_hud(): font = pygame.font.SysFont("arial", 26, bold=True) small = pygame.font.SysFont("arial", 22) health = font.render(f"HP: {int(max(0, player['health']))}", True, (255, 95, 95)) weapon_name = font.render(f"Arma: {player['weapon'].upper()}", True, (255, 255, 255)) ammo = font.render( f"Munición: {player['ammo'][player['weapon']]['mag']} / {player['ammo'][player['weapon']]['reserve']}", True, (255, 255, 255), ) score = font.render(f"Puntuación: {player['score']}", True, (255, 210, 100)) wave_text = font.render(f"Ola: {wave}", True, (170, 245, 170)) screen.blit(health, (30, 30)) screen.blit(weapon_name, (30, 65)) screen.blit(ammo, (30, 100)) screen.blit(score, (30, 135)) screen.blit(wave_text, (30, 170)) cx, cy = WIDTH // 2, HEIGHT // 2 pygame.draw.line(screen, (255, 255, 255), (cx - 12, cy), (cx + 12, cy), 2) pygame.draw.line(screen, (255, 255, 255), (cx, cy - 12), (cx, cy + 12), 2) if player["reload_timer"] > 0: reload_text = font.render("RECARGANDO...", True, (255, 220, 100)) screen.blit(reload_text, (WIDTH // 2 - 120, HEIGHT - 70)) if game_over: big = pygame.font.SysFont("arial", 52, bold=True) txt = big.render("DERROTADO", True, (255, 70, 70)) screen.blit(txt, (WIDTH // 2 - 170, HEIGHT // 2 - 30)) msg = small.render("Presiona R para reiniciar", True, (255, 255, 255)) screen.blit(msg, (WIDTH // 2 - 140, HEIGHT // 2 + 40)) def draw_world(): # Fondo del bosque sky = (75, 120, 150) horizon = (40, 75, 55) ground = (22, 30, 18) pygame.draw.rect(screen, sky, (0, 0, WIDTH, HEIGHT // 2)) pygame.draw.rect(screen, horizon, (0, HEIGHT // 2 - 20, WIDTH, 25)) pygame.draw.rect(screen, ground, (0, HEIGHT // 2, WIDTH, HEIGHT // 2)) # Árboles del fondo (silhouette) for i in range(15): x = (i * 100 + 30) % (WIDTH + 100) y = 250 + (i % 3) * 25 pygame.draw.rect(screen, (20, 30, 18), (x, y + 25, 16, 80)) pygame.draw.circle(screen, (24, 70, 38), (x + 8, y), 28) pygame.draw.circle(screen, (20, 90, 52), (x - 12, y + 10), 24) pygame.draw.circle(screen, (18, 82, 42), (x + 18, y + 8), 26) visible = [] for z in zombies: if not z["alive"]: continue dx = z["x"] - player["x"] dy = z["y"] - player["y"] dist = math.hypot(dx, dy) ang = math.atan2(dy, dx) diff = normalize_angle(ang - player["angle"]) if abs(diff) < FOV * 0.9 and dist < MAX_DEPTH: visible.append((dist, diff, z)) visible.sort(key=lambda v: v[0], reverse=True) # Raycast del mapa for col in range(0, WIDTH, 2): ray_angle = player["angle"] - FOV / 2 + (col / WIDTH) * FOV dist, hit_x, hit_y = cast_ray(ray_angle) corrected = dist * math.cos(ray_angle - player["angle"]) if corrected <= 0.0001: corrected = 0.0001 wall_h = clamp(HEIGHT / corrected, 30, HEIGHT * 2) y = (HEIGHT - wall_h) / 2 shade = clamp(int(255 * (1 - corrected / MAX_DEPTH)), 35, 200) color = (shade, int(shade * 0.8), int(shade * 0.55)) pygame.draw.rect(screen, color, (col, y, 2, wall_h)) # Sprites de zombies for dist, diff, z in visible: sprite_x = WIDTH / 2 + (diff / FOV) * WIDTH size = clamp((HEIGHT / dist) * 0.75, 20, 300) sprite_y = HEIGHT / 2 + size * 0.12 rect = pygame.Rect( sprite_x - size * 0.25, sprite_y - size * 0.9, size * 0.5, size * 1.5, ) body_color = (20, 160, 90) if z["hit_flash"] <= 0 else (255, 110, 110) pygame.draw.rect(screen, body_color, rect) pygame.draw.circle(screen, (20, 20, 20), (int(sprite_x), int(sprite_y - size * 0.7)), int(size * 0.18)) pygame.draw.rect(screen, (255, 60, 60), (rect.x, rect.y - 8, rect.w * (z["hp"] / 100), 5)) def draw_minimap(): map_size = 180 sx = 20 sy = HEIGHT - map_size - 20 pygame.draw.rect(screen, (30, 30, 30), (sx, sy, map_size, map_size), 2) cell = map_size / len(MAP) for j, row in enumerate(MAP): for i, ch in enumerate(row): if ch == "1": pygame.draw.rect(screen, (80, 80, 80), (sx + i * cell, sy + j * cell, cell, cell)) # jugador px = sx + player["x"] * cell py = sy + player["y"] * cell pygame.draw.circle(screen, (80, 200, 255), (int(px), int(py)), 4) # zombies for z in zombies: if z["alive"]: zx = sx + z["x"] * cell zy = sy + z["y"] * cell pygame.draw.circle(screen, (220, 70, 70), (int(zx), int(zy)), 3) def main(): global zombies reset_game() running = True while running: dt = clock.tick(60) / 1000.0 for event in pygame.event.get(): if event.type == pygame.QUIT: running = False if event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE: running = False if event.key == pygame.K_1: player["weapon"] = "pistol" if event.key == pygame.K_2: player["weapon"] = "rifle" if event.key == pygame.K_r: if game_over: reset_game() else: reload_weapon() if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1: fire() keys = pygame.key.get_pressed() if not game_over: update_player(dt, keys) update_zombies(dt) if game_over and keys[pygame.K_r]: reset_game() draw_world() draw_hud() draw_minimap() pygame.display.flip() pygame.quit() if __name__ == "__main__": main()
